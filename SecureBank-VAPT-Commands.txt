-H "Content-Type: application/json" \
-d '{"username":"testuser","email":"testuser@example.com","password":"Test@12345"}'


# 5. LOGIN / JWT TEST
# ============================================================

curl -i -X POST http://127.0.0.1:5000/api/auth/login \
-H "Content-Type: application/json" \
-d '{"username":"testuser","password":"Test@12345"}'

# JWT ko terminal variable mein manually set karna:
TOKEN="PASTE_JWT_HERE"


# 6. CREATE BANK ACCOUNT
# ============================================================

curl -i -X POST http://127.0.0.1:5000/api/accounts/create \
-H "Content-Type: application/json" \
-H "Authorization: Bearer $TOKEN"


# 7. MY ACCOUNT
# ============================================================

curl -i http://127.0.0.1:5000/api/accounts/my-account \
-H "Authorization: Bearer $TOKEN"

# 10. MISSING JWT TEST

# 11. INVALID JWT TEST
# ============================================================

-H "Authorization: Bearer fake.invalid.token"


# 12. MALFORMED AUTH HEADER TEST
# ============================================================

-H "Authorization: InvalidToken"


# 13. SQL INJECTION TEST
# ============================================================

curl -i -X POST http://127.0.0.1:5000/api/auth/login \
-H "Content-Type: application/json" \
-d '{"username":"'\'' OR 1=1 --","password":"test"}'


# Expected:
# 401 Unauthorized
# invalid credentials
#
# This indicates authentication bypass was NOT demonstrated.


# 14. XSS TEST
# ============================================================

curl -i -X POST http://127.0.0.1:5000/api/auth/register \
-H "Content-Type: application/json" \
-d '{"username":"xsstest2026","email":"xsstest2026@example.com","password":"Test@12345<script>alert(1)</script>"}'


# Login with same test password:

curl -i -X POST http://127.0.0.1:5000/api/auth/login \
-H "Content-Type: application/json" \
-d '{"username":"xsstest2026","password":"Test@12345<script>alert(1)</script>"}'


# Set returned JWT locally:
TOKEN_XSS="PASTE_JWT_HERE"


# Check authenticated endpoint:

curl -i http://127.0.0.1:5000/api/accounts/my-account \
-H "Authorization: Bearer $TOKEN_XSS"
